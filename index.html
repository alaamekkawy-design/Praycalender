// استبدل دالة updatePrayerTableDisplay بهذه الدالة الجديدة function updatePrayerTableDisplay(prayerTimes, cityName) { const tbody = document.getElementById('prayerBody'); tbody.innerHTML = ''; // صف واحد فقط للمدينة المختارة const row = tbody.insertRow(); row.insertCell(0).innerText = cityName; if (prayerTimes) { row.insertCell(1).innerHTML = `${prayerTimes.fajr}`; row.insertCell(2).innerHTML = `${prayerTimes.sunrise}`; row.insertCell(3).innerHTML = `${prayerTimes.dhuhr}`; row.insertCell(4).innerHTML = `${prayerTimes.asr}`; row.insertCell(5).innerHTML = `${prayerTimes.maghrib}`; row.insertCell(6).innerHTML = `${prayerTimes.isha}`; } else { const cell = row.insertCell(1); cell.colSpan = 6; cell.innerText = 'خطأ في التحميل'; } }
